<script>
const START=new Date(2026,9,26),END=new Date(2027,0,24),KEY='meta-100-25';
const BASE={f:30,b:10},GOAL={f:100,b:25};
let S={};try{S=JSON.parse(localStorage.getItem(KEY)||'{}')}catch(e){}
const $=id=>document.getElementById(id),DAY=864e5;
const pad=n=>String(n).padStart(2,'0');
const fmt=d=>`${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
const parse=s=>{const[a,b,c]=s.split('-').map(Number);return new Date(a,b-1,c)};
const idxOf=d=>Math.round((d-START)/DAY);
const typeOf=i=>i%2===0?'f':'b';
const save=()=>localStorage.setItem(KEY,JSON.stringify(S));
const now=new Date(),today=new Date(now.getFullYear(),now.getMonth(),now.getDate());
let cur=today<START?START:today>END?END:today;

function best(t){let m=BASE[t];for(const k in S){const v=+S[k].best;if(v&&typeOf(idxOf(parse(k)))===t&&v>m)m=v}return m}
function rx(i){
  const t=typeOf(i),wk=Math.floor(i/7)+1,M=best(t),p=f=>Math.max(1,Math.round(M*f)),b=t==='b';
  if(i<0||i>90)return'Fora do período do plano.';
  if(i===90)return`🏁 TESTE FINAL: aqueça e faça 1 série máxima de flexão (meta ${GOAL.f}).`;
  if(i===89)return`🏁 TESTE FINAL: aqueça e faça 1 série máxima de barra (meta ${GOAL.b}).`;
  if([4,8,12].includes(wk)){
    return(i-(wk-1)*7<2?`Teste: 1 série máxima no padrão estrito (registre abaixo), depois `:'')+`2 × ${p(.4)} reps leves.`}
  if(wk===13)return`Polimento: 3 × ${p(.5)} reps, sem ir perto da falha.`;
  if(wk<=3){const r=(b?[120,105,90]:[90,75,60])[wk-1];return`${b?5:6} × ${p(.6)} reps, descanso ${r}s.`}
  if(wk<=7){const k=wk-5;return`EMOM de ${(b?8:10)+2*k} min com ${Math.max(2,p(.4))} reps por minuto, + 1 série a 2 reps da falha.`}
  const f=[.75,.8,.85][wk-9];
  return`4 × ${p(f)} reps, descanso 2 min, `+(b?`+ 3 descidas excêntricas de 5s.`:`+ 1 série máxima.`)}
const CORE={f:'Core (2–3 voltas): prancha 40s · hollow hold 20s · dead bug 10/lado.',
 b:'Core (2–3 voltas): elevação de joelhos na barra 8–12 · L-sit grupado 15s · prancha lateral 30s/lado.'};

function streak(){let d=new Date(today);if(!S[fmt(d)]?.done)d=new Date(d-DAY);let n=0;
  while(S[fmt(d)]?.done){n++;d=new Date(d.getFullYear(),d.getMonth(),d.getDate()-1)}return n}
function render(){
  const left=Math.ceil((END-today)/DAY);
  $('cd').textContent=left>0?left:left===0?'HOJE':'✓';
  $('cdl').textContent=left>0?'dias p/ 24/jan':left===0?'dia do teste final':'prazo encerrado';
  $('st').textContent=streak();
  for(const[t,id,bid]of[['f','fl','fb'],['b','ba','bb']]){
    const c=best(t),pc=Math.min(100,Math.max(0,(c-BASE[t])/(GOAL[t]-BASE[t])*100));
    $(id).textContent=`${BASE[t]} → ${GOAL[t]} · atual ${c} (${Math.round(pc)}%)`;$(bid).style.width=pc+'%'}
  const i=idxOf(cur),t=typeOf(i),k=fmt(cur),e=S[k]||{};
  $('dt').value=k;
  $('type').textContent=t==='f'?'FLEXÃO':'BARRA';
  $('wk').textContent=`Semana ${Math.floor(i/7)+1} · dia ${i+1}`;
  $('rx').textContent=rx(i);$('core').textContent=CORE[t];
  $('ck').textContent=e.done?'✅ Treino concluído (toque para desfazer)':'Marcar treino como concluído';
  $('ck').className=e.done?'done':'';
  $('best').value=e.best||'';$('notes').value=e.notes||'';
}
function upd(f){const k=fmt(cur);S[k]=Object.assign(S[k]||{},f);save();render()}
$('ck').onclick=()=>upd({done:!(S[fmt(cur)]||{}).done});
$('best').onchange=e=>upd({best:+e.target.value||0});
$('notes').oninput=e=>{const k=fmt(cur);S[k]=Object.assign(S[k]||{},{notes:e.target.value});save()};
$('dt').onchange=e=>{const d=parse(e.target.value);if(d>=START&&d<=END){cur=d;render()}};
$('prev').onclick=()=>{const d=new Date(cur.getFullYear(),cur.getMonth(),cur.getDate()-1);if(d>=START){cur=d;render()}};
$('next').onclick=()=>{const d=new Date(cur.getFullYear(),cur.getMonth(),cur.getDate()+1);if(d<=END){cur=d;render()}};
$('exp').onclick=()=>navigator.clipboard.writeText(JSON.stringify(S)).then(()=>alert('Backup copiado!'));
render();
if('serviceWorker'in navigator)navigator.serviceWorker.register('service-worker.js');
</script></body></html>
